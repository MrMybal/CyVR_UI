cmake_minimum_required(VERSION 3.22)
project(CyVR_UI VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options("-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=/cyvr" "-fmacro-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=/cyvr")
  if(ANDROID)
    add_compile_options("-ffile-prefix-map=${ANDROID_NDK}=/android-ndk" "-fmacro-prefix-map=${ANDROID_NDK}=/android-ndk")
  endif()
endif()
option(CYVR_BUILD_TESTS "Build portable UI tests" ON)
option(CYVR_ENABLE_IMMERSIVE "Include the optional OpenXR host alongside native 2D" ON)
add_library(cyvr_ui STATIC src/core.cpp src/ui.cpp src/spatial.cpp src/locale.cpp)
target_include_directories(cyvr_ui PUBLIC include)
if(MSVC)
  target_compile_options(cyvr_ui PUBLIC /utf-8 /W4 /permissive-)
else()
  target_compile_options(cyvr_ui PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CYVR_BUILD_TESTS AND NOT ANDROID)
  enable_testing()
  add_executable(cyvr_tests tests/core_tests.cpp)
  target_link_libraries(cyvr_tests PRIVATE cyvr_ui)
  add_test(NAME ui_core COMMAND cyvr_tests)
endif()
find_program(CYVR_GLSLC NAMES glslc REQUIRED)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/assets")
foreach(shader ui.vert ui.frag spatial.vert spatial.frag environment.vert environment.frag glass.frag)
  add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/assets/${shader}.spv"
    COMMAND "${CYVR_GLSLC}" --target-env=vulkan1.1 -O "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}" -o "${CMAKE_BINARY_DIR}/assets/${shader}.spv"
    DEPENDS "shaders/${shader}")
endforeach()
configure_file(assets/font_atlas.bin assets/font_atlas.bin COPYONLY)
configure_file(assets/hand_studio.bin assets/hand_studio.bin COPYONLY)
add_custom_target(cyvr_assets ALL DEPENDS "${CMAKE_BINARY_DIR}/assets/ui.vert.spv" "${CMAKE_BINARY_DIR}/assets/ui.frag.spv" "${CMAKE_BINARY_DIR}/assets/spatial.vert.spv" "${CMAKE_BINARY_DIR}/assets/spatial.frag.spv" "${CMAKE_BINARY_DIR}/assets/environment.vert.spv" "${CMAKE_BINARY_DIR}/assets/environment.frag.spv" "${CMAKE_BINARY_DIR}/assets/glass.frag.spv")
add_library(cyvr_gpu STATIC src/gpu.cpp)
target_link_libraries(cyvr_gpu PUBLIC cyvr_ui)
target_include_directories(cyvr_gpu PUBLIC third_party/vulkan/include)
if(ANDROID)
  set(NATIVE_GLUE "${ANDROID_NDK}/sources/android/native_app_glue")
  add_library(native_glue STATIC "${NATIVE_GLUE}/android_native_app_glue.c")
  target_include_directories(native_glue PUBLIC "${NATIVE_GLUE}")
  add_library(cyvr_android STATIC src/android_host.cpp)
  target_link_libraries(cyvr_android PUBLIC cyvr_gpu native_glue android log dl)
  target_compile_definitions(cyvr_android PUBLIC CYVR_ENABLE_IMMERSIVE=$<BOOL:${CYVR_ENABLE_IMMERSIVE}>)
  if(CYVR_ENABLE_IMMERSIVE)
    add_library(openxr_loader SHARED IMPORTED)
    set_target_properties(openxr_loader PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/.deps/openxr/prefab/modules/openxr_loader/libs/android.arm64-v8a/libopenxr_loader.so")
    target_sources(cyvr_android PRIVATE src/xr_app.cpp)
    target_include_directories(cyvr_android PRIVATE third_party/openxr/include)
    target_compile_definitions(cyvr_android PRIVATE XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_VULKAN)
    target_link_libraries(cyvr_android PUBLIC openxr_loader)
  endif()
  add_library(cyvr SHARED src/android_main.cpp)
  target_link_libraries(cyvr PRIVATE cyvr_android)
  target_link_options(cyvr PRIVATE -uANativeActivity_onCreate -Wl,-z,max-page-size=16384)
  add_dependencies(cyvr cyvr_assets)
elseif(WIN32)
  add_executable(CyVR_UI src/desktop.cpp)
  target_link_libraries(CyVR_UI PRIVATE cyvr_gpu user32 gdi32 psapi)
  add_dependencies(CyVR_UI cyvr_assets)
endif()
