cmake_minimum_required(VERSION 3.22)
project(CyVR_UI VERSION 0.8.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options("-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=/cyvr" "-fmacro-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=/cyvr")
  if(ANDROID)
    add_compile_options("-ffile-prefix-map=${ANDROID_NDK}=/android-ndk" "-fmacro-prefix-map=${ANDROID_NDK}=/android-ndk")
  endif()
endif()
option(CYVR_BUILD_TESTS "Build portable UI tests" ON)
option(CYVR_BUILD_GPU_TESTS "Build Vulkan rendering regression checks" OFF)
option(CYVR_ENABLE_IMMERSIVE "Include the optional OpenXR host alongside native 2D" ON)
option(CYVR_BUILD_REFERENCE "Build the standalone reference application" ON)
set(CYVR_SCENES_DEFAULT OFF)
if(CYVR_BUILD_REFERENCE AND CYVR_ENABLE_IMMERSIVE)
  set(CYVR_SCENES_DEFAULT ON)
endif()
option(CYVR_INCLUDE_SCENES "Bundle optional CC0 photographic environments" ${CYVR_SCENES_DEFAULT})
option(CYVR_INCLUDE_PBR "Bundle original PBR materials and baked HDR probes" ${CYVR_SCENES_DEFAULT})
add_library(cyvr_ui STATIC src/core.cpp src/ui.cpp src/spatial.cpp src/scene.cpp src/pbr.cpp src/scene_lighting.cpp src/locale.cpp)
target_include_directories(cyvr_ui PUBLIC include)
if(MSVC)
  target_compile_options(cyvr_ui PUBLIC /utf-8 /W4 /permissive-)
else()
  target_compile_options(cyvr_ui PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CYVR_BUILD_TESTS AND NOT ANDROID)
  enable_testing()
  add_executable(cyvr_tests tests/core_tests.cpp)
  target_link_libraries(cyvr_tests PRIVATE cyvr_ui)
  target_include_directories(cyvr_tests PRIVATE third_party/openxr/include)
  add_test(NAME ui_core COMMAND cyvr_tests)
endif()
option(CYVR_BUILD_SCENE_BAKER "Build the offline triangle-based lighting authoring tool" OFF)
if(CYVR_BUILD_SCENE_BAKER AND NOT ANDROID)
  add_executable(cyvr_bake_lighting tools/bake_scene_lighting.cpp)
  target_link_libraries(cyvr_bake_lighting PRIVATE cyvr_ui)
  if(CYVR_BUILD_TESTS)
    add_test(NAME scene_baker COMMAND cyvr_bake_lighting --self-test)
  endif()
endif()
find_program(CYVR_GLSLC NAMES glslc REQUIRED)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/assets")
foreach(shader ui.vert ui.frag spatial.vert spatial.frag environment.vert environment.frag glass.frag scene.vert scene.frag scene_blur.frag)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/assets/${shader}.spv"
    COMMAND "${CYVR_GLSLC}" --target-env=vulkan1.1 -O "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}" -o "${CMAKE_CURRENT_BINARY_DIR}/assets/${shader}.spv"
    DEPENDS "shaders/${shader}")
endforeach()
configure_file(assets/font_atlas.bin "${CMAKE_CURRENT_BINARY_DIR}/assets/font_atlas.bin" COPYONLY)
configure_file(assets/hand_studio.bin "${CMAKE_CURRENT_BINARY_DIR}/assets/hand_studio.bin" COPYONLY)
if(CYVR_INCLUDE_SCENES)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/assets/scenes")
  foreach(scene lebombo venice_sunset)
    configure_file("assets/scenes/${scene}.jpg" "${CMAKE_CURRENT_BINARY_DIR}/assets/scenes/${scene}.jpg" COPYONLY)
  endforeach()
endif()
if(CYVR_INCLUDE_PBR)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/assets/pbr")
  foreach(texture basecolor normal-rough-metal brdf lounge observatory terrace lighting-lounge lighting-observatory lighting-terrace)
    configure_file("assets/pbr/${texture}.bin" "${CMAKE_CURRENT_BINARY_DIR}/assets/pbr/${texture}.bin" COPYONLY)
  endforeach()
endif()
add_custom_target(cyvr_assets ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/assets/ui.vert.spv" "${CMAKE_CURRENT_BINARY_DIR}/assets/ui.frag.spv" "${CMAKE_CURRENT_BINARY_DIR}/assets/spatial.vert.spv" "${CMAKE_CURRENT_BINARY_DIR}/assets/spatial.frag.spv" "${CMAKE_CURRENT_BINARY_DIR}/assets/environment.vert.spv" "${CMAKE_CURRENT_BINARY_DIR}/assets/environment.frag.spv" "${CMAKE_CURRENT_BINARY_DIR}/assets/glass.frag.spv" "${CMAKE_CURRENT_BINARY_DIR}/assets/scene.vert.spv" "${CMAKE_CURRENT_BINARY_DIR}/assets/scene.frag.spv" "${CMAKE_CURRENT_BINARY_DIR}/assets/scene_blur.frag.spv")
add_library(cyvr_gpu STATIC src/gpu.cpp)
target_link_libraries(cyvr_gpu PUBLIC cyvr_ui)
target_include_directories(cyvr_gpu PRIVATE third_party/stb)
if(NOT ANDROID)
  target_include_directories(cyvr_gpu PUBLIC third_party/vulkan/include)
endif()
if(CYVR_BUILD_GPU_TESTS AND CYVR_BUILD_TESTS AND WIN32 AND NOT ANDROID)
  add_executable(cyvr_render_tests tests/render_tests.cpp)
  target_link_libraries(cyvr_render_tests PRIVATE cyvr_gpu)
  add_dependencies(cyvr_render_tests cyvr_assets)
  add_test(NAME gpu_rendering COMMAND cyvr_render_tests "${CMAKE_CURRENT_BINARY_DIR}/assets")
endif()
if(ANDROID)
  set(NATIVE_GLUE "${ANDROID_NDK}/sources/android/native_app_glue")
  add_library(native_glue STATIC "${NATIVE_GLUE}/android_native_app_glue.c")
  target_include_directories(native_glue PUBLIC "${NATIVE_GLUE}")
  add_library(cyvr_android STATIC src/android_host.cpp)
  target_link_libraries(cyvr_android PUBLIC cyvr_gpu native_glue android log dl)
  target_compile_definitions(cyvr_android PUBLIC CYVR_ENABLE_IMMERSIVE=$<BOOL:${CYVR_ENABLE_IMMERSIVE}>)
  if(CYVR_ENABLE_IMMERSIVE)
    add_library(openxr_loader SHARED IMPORTED)
    set_target_properties(openxr_loader PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/.deps/openxr/prefab/modules/openxr_loader/libs/android.arm64-v8a/libopenxr_loader.so")
    target_sources(cyvr_android PRIVATE src/xr_app.cpp)
    target_include_directories(cyvr_android PRIVATE third_party/openxr/include)
    target_compile_definitions(cyvr_android PRIVATE XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_VULKAN)
    target_link_libraries(cyvr_android PUBLIC openxr_loader)
  endif()
  if(CYVR_BUILD_REFERENCE)
    add_library(cyvr SHARED src/android_main.cpp)
    target_link_libraries(cyvr PRIVATE cyvr_android)
    target_link_options(cyvr PRIVATE -uANativeActivity_onCreate -Wl,-z,max-page-size=16384)
    add_dependencies(cyvr cyvr_assets)
  endif()
elseif(WIN32 AND CYVR_BUILD_REFERENCE)
  add_executable(CyVR_UI src/desktop.cpp)
  target_link_libraries(CyVR_UI PRIVATE cyvr_gpu user32 gdi32 psapi)
  add_dependencies(CyVR_UI cyvr_assets)
endif()
